function getConsultas() {
  const data = localStorage.getItem('lcf_consultas');
  return data ? JSON.parse(data) : [];
}
function saveConsulta(consulta) {
  const consultas = getConsultas();
  consultas.unshift(consulta);
  localStorage.setItem('lcf_consultas', JSON.stringify(consultas));
}

    tipoEvento: formData.tipo,

}

function openForm() {
  const s = SALONES.find(s => s.id === currentSalonId);
  const badge = document.getElementById('form-salon-badge');
  badge.style.background = s.colorPale;
  badge.style.color = s.color;
  badge.style.border = `1.5px solid ${s.colorPale.replace('.1','0.4')}`;
  badge.innerHTML = `📍 ${s.nombre} · ${s.tipo}`;
  const btn = document.getElementById('form-submit-btn');
  btn.style.background = s.btnGradient;
  const today = new Date().toISOString().split('T')[0];
  document.getElementById('f-fecha').min = today;
  clearFormErrors();
  document.getElementById('form-overlay').classList.add('open');
  document.body.style.overflow = 'hidden';
}
function closeForm() {
  document.getElementById('form-overlay').classList.remove('open');
  document.body.style.overflow = '';
}
function clearFormErrors() {
  ['fg-nombre','fg-telefono','fg-fecha','fg-invitados','fg-tipo'].forEach(id => {
    document.getElementById(id).classList.remove('error');
  });
}
function validateForm() {
  let valid = true;
  const checks = [
    {id:'fg-nombre', val: document.getElementById('f-nombre').value.trim()},
    {id:'fg-telefono', val: document.getElementById('f-telefono').value.trim()},
    {id:'fg-fecha', val: document.getElementById('f-fecha').value},
    {id:'fg-invitados', val: document.getElementById('f-invitados').value},
    {id:'fg-tipo', val: document.getElementById('f-tipo').value},
  ];
  clearFormErrors();
  checks.forEach(c => {
    if (!c.val) {
      document.getElementById(c.id).classList.add('error');
      valid = false;
    }
  });
  return valid;
}
function getFormData() {
  return {
    nombre: document.getElementById('f-nombre').value.trim(),
    telefono: document.getElementById('f-telefono').value.trim(),
    fecha: document.getElementById('f-fecha').value,
    horario: document.getElementById('f-horario').value,
    invitados: document.getElementById('f-invitados').value,
    tipo: document.getElementById('f-tipo').value,
    mensaje: document.getElementById('f-mensaje').value.trim()
  };
}
function resetForm() {
  ['f-nombre','f-telefono','f-invitados','f-mensaje'].forEach(id => document.getElementById(id).value = '');
  document.getElementById('f-fecha').value = '';
  document.getElementById('f-horario').value = '19:00';
  document.getElementById('f-tipo').value = '';
}
function enviarConsulta() {
  if (!validateForm()) return;
  const salon = SALONES.find(s => s.id === currentSalonId);
  const formData = getFormData();
  const consulta = crearConsulta(currentSalonId, formData);
  saveConsulta(consulta);
  closeForm();
  abrirWhatsApp(salon, formData);
  document.getElementById('success-msg').textContent = `Tu consulta para ${salon.nombre} fue guardada. WhatsApp se abrirá con tu mensaje listo para enviar.`;
  document.getElementById('success-overlay').classList.add('open');
  resetForm();
}
function closeSuccess() {
  document.getElementById('success-overlay').classList.remove('open');
}

function filterAdmin(btn, filter) {
  document.querySelectorAll('.af').forEach(b => b.classList.remove('active'));
  btn.classList.add('active');
  adminFilter = filter;
  renderAdminConsultas();
}
function renderAdminConsultas() {
  const todas = getConsultas();
  let items = todas;
  if (adminFilter === 'nueva') items = todas.filter(c => c.estado === 'nueva');
  else if (['aserran','chiquiclub','acuarelas','aserrin-aserran'].includes(adminFilter)) items = todas.filter(c => c.salonId === adminFilter);
  const list = document.getElementById('admin-consult-list');
  document.getElementById('kpi-consultas').textContent = todas.length;
  document.getElementById('kpi-reservadas').textContent = todas.filter(c => c.estado === 'reservada').length;
  document.getElementById('kpi-nuevas').textContent = todas.filter(c => c.estado === 'nueva').length;
  if (items.length === 0) {
    list.innerHTML = `<div class="empty-state"><span>📭</span><p>No hay consultas${adminFilter !== 'todas' ? ' con este filtro' : ''} todavía.</p></div>`;
    return;
  }
  list.innerHTML = items.map(c => {
    const salon = SALONES.find(s => s.id === c.salonId);
    const colorPale = salon ? salon.colorPale : 'rgba(124,140,255,.1)';
    const color = salon ? salon.color : '#7C8CFF';
    const fecha = c.fechaEvento ? new Date(c.fechaEvento+'T00:00:00').toLocaleDateString('es-UY',{day:'numeric',month:'short',year:'numeric'}) : '—';
    const creado = new Date(c.createdAt).toLocaleDateString('es-UY',{day:'numeric',month:'short',hour:'2-digit',minute:'2-digit'});
    return `<div class="consult-item" onclick="toggleDetail('${c.id}')">
      <div class="ci-avatar av-web" style="background:${colorPale};color:${color}">${c.salonNombre.substring(0,2).toUpperCase()}</div>
      <div class="ci-info">
        <div class="ci-name">${c.nombreCliente}<span class="ci-salon-badge" style="background:${colorPale};color:${color}">${c.salonNombre}</span></div>
        <div class="ci-det">${c.tipoEvento} · ${c.cantidadInvitados} personas · ${fecha}</div>
        ${c.mensaje ? `<div class="ci-msg">"${c.mensaje.substring(0,50)}${c.mensaje.length>50?'...':''}"</div>` : ''}
      </div>
      <div class="ci-meta">
        <span class="status-badge st-${c.estado}">${c.estado}</span>
        <div class="ci-time">${creado}</div>
      </div>
    </div>
    <div class="consult-detail" id="detail-${c.id}">
      <div class="cd-grid">
        <div class="cd-field"><div class="cd-label">Teléfono</div><div class="cd-value">${c.telefono}</div></div>
        <div class="cd-field"><div class="cd-label">Fecha evento</div><div class="cd-value">${fecha}</div></div>
        <div class="cd-field"><div class="cd-label">Horario</div><div class="cd-value">${c.horario||'—'}</div></div>
        <div class="cd-field"><div class="cd-label">Estado</div><div class="cd-value"><span class="status-badge st-${c.estado}">${c.estado}</span></div></div>
      </div>
      ${c.mensaje ? `<div class="cd-msg-box"><div class="cd-label" style="margin-bottom:4px">Mensaje</div><div class="cd-value">${c.mensaje}</div></div>` : ''}
      <div class="cd-actions">
        <button class="cd-btn cd-btn-confirm" onclick="cambiarEstado('${c.id}','reservada');event.stopPropagation()">✓ Reservada</button>
        <button class="cd-btn cd-btn-wa" onclick="contactarWA('${c.id}');event.stopPropagation()">💬 WhatsApp</button>
        <button class="cd-btn cd-btn-reject" onclick="cambiarEstado('${c.id}','perdida');event.stopPropagation()">✗ Perdida</button>
      </div>
    </div>`;

}

function cambiarEstado(id, nuevoEstado) {
  const consultas = getConsultas();
  const idx = consultas.findIndex(c => c.id === id);
  if (idx > -1) { consultas[idx].estado = nuevoEstado; localStorage.setItem('lcf_consultas', JSON.stringify(consultas)); renderAdminConsultas(); }
}
function contactarWA(id) {
  const c = getConsultas().find(c => c.id === id);
  const salon = SALONES.find(s => s.id === c.salonId);
  const msg = encodeURIComponent(`Hola ${c.nombreCliente}, te contactamos desde *La Casa de las Fiestas* por tu consulta para el salón *${c.salonNombre}*.\n\nTu evento: ${c.tipoEvento} · ${c.fechaEvento} · ${c.cantidadInvitados} personas.\n\n¿Podemos confirmar la disponibilidad? 🎉`);
  window.open(`https://wa.me/${salon.whatsapp}?text=${msg}`, '_blank');
}
